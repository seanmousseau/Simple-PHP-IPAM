## Task 1: TOTP Library Evaluation and Composer Addition

> **No UI/UX work in this task.**

**Context:** #418 requires TOTP (RFC 6238). The project's dep policy requires: narrow purpose, mature, minimal transitive deps, MIT/BSD/Apache license, no framework dependencies.

**Chosen library: `robthree/twofactorauth`**
- Version constraint: `^2.1`
- License: MIT
- Zero runtime transitive dependencies
- Pure-PHP TOTP/HOTP implementation, PHP 8.0+ native
- Used in ~3 000 packages on Packagist
- Handles: secret generation, QR URI generation, code verification with configurable leeway
- Does NOT generate QR images — we'll render the `otpauth://` URI client-side via a JavaScript QR library (endroid/qr-code is server-side and too heavy; we vendor a tiny JS lib instead)

**Chosen JS QR library: `qrcode.js` (davidshimjs)**
- Single file, ~8KB minified
- MIT license, zero deps
- Self-hosted at `Simple-PHP-IPAM/assets/vendor/qrcode.min.js`
- Renders QR to `<canvas>` from any string

```php
// --- TOTP 2FA helpers ---

function ipam_totp_tfa(): \RobThree\Auth\TwoFactorAuth {
    static $tfa = null;
    if ($tfa === null) {
        $tfa = new \RobThree\Auth\TwoFactorAuth(null, 6, 30, 'sha1');
    }
    return $tfa;
}

function ipam_totp_generate_secret(): string {
    return ipam_totp_tfa()->createSecret(160); // 160 bits = 32 base32 chars
}

function ipam_totp_get_uri(string $secret, string $issuer, string $accountName): string {
    return ipam_totp_tfa()->getQRCodeUrl($issuer, $accountName, $secret);
}

function ipam_totp_verify(string $secret, string $code, int $discrepancy = 1): bool {
    return ipam_totp_tfa()->verifyCode($secret, $code, $discrepancy);
}

function ipam_totp_encrypt_secret(string $secret, string $key): string {
    $iv = random_bytes(16);
    $enc = openssl_encrypt($secret, 'aes-256-cbc', hash('sha256', $key, true), OPENSSL_RAW_DATA, $iv);
    return base64_encode($iv . $enc);
}

function ipam_totp_decrypt_secret(string $encSecret, string $key): string {
    $raw = base64_decode($encSecret);
    $iv = substr($raw, 0, 16);
    $enc = substr($raw, 16);
    return openssl_decrypt($enc, 'aes-256-cbc', hash('sha256', $key, true), OPENSSL_RAW_DATA, $iv);
}

function ipam_totp_generate_backup_codes(int $count = 8): array {
    $codes = [];
    for ($i = 0; $i < $count; $i++) {
        $codes[] = strtoupper(bin2hex(random_bytes(4))) . '-' . strtoupper(bin2hex(random_bytes(4)));
    }
    return $codes;
}

function ipam_totp_save_backup_codes(PDO $db, int $userId, array $codes): void {
    $db->prepare("DELETE FROM totp_backup_codes WHERE user_id = :uid")->execute([':uid' => $userId]);
    $stmt = $db->prepare("INSERT INTO totp_backup_codes (user_id, code_hash) VALUES (:uid, :hash)");
    foreach ($codes as $code) {
        $stmt->execute([':uid' => $userId, ':hash' => password_hash($code, PASSWORD_DEFAULT)]);
    }
}

function ipam_totp_verify_backup_code(PDO $db, int $userId, string $code): bool {
    $rows = $db->prepare("SELECT id, code_hash FROM totp_backup_codes WHERE user_id = :uid AND used_at IS NULL")
               ->execute([':uid' => $userId]);
    $rows = $db->prepare("SELECT id, code_hash FROM totp_backup_codes WHERE user_id = :uid AND used_at IS NULL");
    $rows->execute([':uid' => $userId]);
    foreach ($rows->fetchAll(PDO::FETCH_ASSOC) as $row) {
        if (password_verify($code, $row['code_hash'])) {
            $db->prepare("UPDATE totp_backup_codes SET used_at = datetime('now') WHERE id = :id")
               ->execute([':id' => $row['id']]);
            return true;
        }
    }
    return false;
}

// --- Rate limiting helpers ---

function ipam_rate_limit_window_key(string $bucketKey, int $windowSeconds, ?int $ts = null): string {
    $t = $ts ?? time();
    $window = (int)floor($t / $windowSeconds) * $windowSeconds;
    return $bucketKey . ':' . $window;
}

function ipam_rate_limit_check(PDO $db, string $bucketKey, int $windowSeconds, int $maxRequests, ?int $ts = null): bool {
    $windowKey = ipam_rate_limit_window_key($bucketKey, $windowSeconds, $ts);
    $db->exec("DELETE FROM rate_limit_buckets WHERE window_start < datetime('now', '-" . ($windowSeconds * 2) . " seconds')");
    $db->prepare("INSERT INTO rate_limit_buckets (bucket_key, window_start, count) VALUES (:k, :w, 1)
                  ON CONFLICT(bucket_key, window_start) DO UPDATE SET count = count + 1")
       ->execute([':k' => $bucketKey, ':w' => $windowKey]);
    $row = $db->prepare("SELECT count FROM rate_limit_buckets WHERE bucket_key = :k AND window_start = :w");
    $row->execute([':k' => $bucketKey, ':w' => $windowKey]);
    $count = (int)($row->fetchColumn() ?: 0);
    return $count <= $maxRequests;
}
```

> **Note on TOTP secret encryption key:** Use `$GLOBALS['config']['app_secret'] ?? 'default-insecure-key'`. The `config.php` template must document `app_secret` as a required field for 2FA. See Task 4 for config key addition.

```php
// --- Session hardening ---

function ipam_session_enforce_absolute_lifetime(): void {
    global $config;
    $lifetime = (int)($config['session']['absolute_lifetime_minutes'] ?? 480); // 8 hours default
    if (isset($_SESSION['_abs_expires']) && time() > $_SESSION['_abs_expires']) {
        session_destroy();
        session_start();
        return;
    }
    if (!isset($_SESSION['_abs_expires'])) {
        $_SESSION['_abs_expires'] = time() + ($lifetime * 60);
    }
}

function ipam_rotate_session(): void {
    session_regenerate_id(true);
    // Reset absolute expiry on rotation
    global $config;
    $lifetime = (int)($config['session']['absolute_lifetime_minutes'] ?? 480);
    $_SESSION['_abs_expires'] = time() + ($lifetime * 60);
}
```

Call `ipam_session_enforce_absolute_lifetime()` in `init.php` after `session_start()` and after `require_login()`.

- [ ] **Step 2: Add lockout helpers to lib.php**

```php
// --- Account lockout ---

function ipam_check_lockout(PDO $db, int $userId): bool {
    $row = $db->prepare("SELECT locked_until FROM users WHERE id = :id");
    $row->execute([':id' => $userId]);
    $lockedUntil = $row->fetchColumn();
    if ($lockedUntil && strtotime($lockedUntil) > time()) {
        return true; // still locked
    }
    return false;
}

function ipam_record_auth_failure(PDO $db, int $userId, string $reason = 'failed_login'): void {
    global $config;
    $threshold = (int)($config['auth']['lockout_after_failures'] ?? 10);
    $lockMinutes = (int)($config['auth']['lockout_duration_minutes'] ?? 30);

    $db->prepare("UPDATE users SET failed_auth_count = failed_auth_count + 1 WHERE id = :id")
       ->execute([':id' => $userId]);

    $row = $db->prepare("SELECT failed_auth_count FROM users WHERE id = :id");
    $row->execute([':id' => $userId]);
    $count = (int)$row->fetchColumn();

    if ($count >= $threshold) {
        $until = date('Y-m-d H:i:s', time() + ($lockMinutes * 60));
        $db->prepare("UPDATE users SET locked_until = :until, lock_reason = :reason WHERE id = :id")
           ->execute([':until' => $until, ':reason' => $reason, ':id' => $userId]);
    }
}

function ipam_clear_auth_failures(PDO $db, int $userId): void {
    $db->prepare("UPDATE users SET failed_auth_count = 0, locked_until = NULL, lock_reason = NULL WHERE id = :id")
       ->execute([':id' => $userId]);
}

function ipam_admin_unlock_user(PDO $db, int $userId, int $actorId, string $actorName): void {
    ipam_clear_auth_failures($db, $userId);
    audit($db, 'user.unlock', 'user', $userId, "Unlocked by admin user ID $actorId ($actorName)");
}
```

```php
            : ipam_totp_verify($secret, $code);

        if ($ok) {

            unset($_SESSION['totp_pending_uid']);
            login_user($pendingUid, $user['username'], $user['role'], $db);

            audit($db, 'auth.totp_success', 'user', $pendingUid, 'TOTP verified');
            header('Location: dashboard.php');
            exit;
        } else {
            ipam_record_auth_failure($db, $pendingUid, 'failed_2fa');
            $error = 'Invalid code. Please try again.';
        }
```
