function ipam_restore_stage_uploaded_file(string $src, string $dst): bool

function ipam_mark_stale_addresses(PDO $db, int $subnetId, int $missThreshold = 3): int
{
    // Defensive clamp — caller threshold comes from a tenant setting that
    // operators can mis-edit. 0/negative produces nonsense SQL; very large
    // values fan out scan_results reads. Bound to [1, 50]. (#1162, F-S2-05)
    if ($missThreshold < 1)  $missThreshold = 1;

    // Reserved IPs (network, IPv4 broadcast) are excluded from stale marking so

function ipam_dhcp_normalize_hostname(string $raw): ?string

    $raw = trim($raw);

```
/**
 * Resolve mysql/pgsql connection parameters for the CLI restore script.
 * Parses $gConf['db_dsn'] when present; falls back to discrete keys. (#1177)
 *
 * @param array<string,mixed> $gConf
 * @return array{driver:string,host:string,port:string,dbname:string,user:string,pass:string}
 */
function ipam_restore_resolve_db_conn(array $gConf): array
{
    $dsn = to_str($gConf['db_dsn'] ?? '');
    $driver = '';
    $host = $port = $dbname = '';

    if ($dsn !== '') {
        [$driver, $rest] = array_pad(explode(':', $dsn, 2), 2, '');
        foreach (explode(';', $rest) as $kv) {
            [$k, $v] = array_pad(explode('=', $kv, 2), 2, '');
            $k = trim($k); $v = trim($v);
            if      ($k === 'host')   $host   = $v;
            elseif  ($k === 'port')   $port   = $v;
            elseif  ($k === 'dbname') $dbname = $v;
        }
    }

    if ($host   === '') $host   = to_str($gConf['db_host'] ?? '127.0.0.1');
    if ($port   === '') $port   = to_str($gConf['db_port'] ?? ($driver === 'pgsql' ? '5432' : '3306'));
    if ($dbname === '') $dbname = to_str($gConf['db_name'] ?? 'ipam');

    return [
        'driver' => $driver,
        'host'   => $host,
        'port'   => $port,
        'dbname' => $dbname,
        'user'   => to_str($gConf['db_user'] ?? ($driver === 'pgsql' ? 'postgres' : 'root')),
        'pass'   => to_str($gConf['db_pass'] ?? ''),
    ];
}
```
