```php
/**
 * #930 (C9): run a query and throw a RuntimeException if it returns false.
 *
 * Collapses the `$db->query(...) ?: throw new RuntimeException(...)` idiom
 * repeated 30+ times across migration closures into one named call that
 * also surfaces the failing SQL in the message.
 */
function ipam_query_or_throw(PDO $db, string $sql): PDOStatement
{
    $st = $db->query($sql);
    if ($st === false) {
        $err = $db->errorInfo();
        throw new RuntimeException(
            'Migration query failed: ' . ($err[2] ?? 'unknown') . ' — SQL: ' . $sql
        );
    }
    return $st;
}
```

```php
/**
 * #926 (C7): run engine-specific DDL inside a migration closure.
 *
 * Replaces the copy-pasted three-arm `if (driver === 'sqlite') ... elseif`
 * dispatch repeated in every post-v2.9.0 migration.

 * @param array{sqlite?:string,mysql?:string,pgsql?:string} $ddlByEngine

function migration_create_table(PDO $db, array $ddlByEngine): void
```
